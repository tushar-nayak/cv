\cvProject{Student Researcher\textnormal{, Computational Engineering \& Robotics Lab, Carnegie Mellon University}}{Aug 2024 -- Jul 2026}
{\small\textit{Real-Time 3D Vascular Deformation Estimation from Monocular Fluoroscopy for Endovascular Robotic Surgery}}\par\vspace{2pt}
\begin{cvitems}
  \item Developed {\textsc{FluoroGRID}}, a multi-modal deep learning framework combining fluoroscopic input feed, with preoperative vascular geometry, sparse control points, and acquisition metadata to predict 3D deformation.
  \item Designed a quad-encoder architecture with image features, ordered-chain graph features, sampled mesh geometry, and camera metadata with differentiable projection and  physics-based geometric regularizers.
  \item Generated over 370,000 synthetic dataset with paired 3D--2D examples across views and deformation conditions.
\item Achieved real-time inference with sub-millimeter reconstruction precision. Presented initial results at ELSR IROS 2026.
\end{cvitems}

\cvProject{Research Assistant\textnormal{, The $\forall$ Lab \& Image Science Lab, Carnegie Mellon University}}{Feb 2025 -- Aug 2025}
{\small\textit{Longitudinal Glioblastoma Evolution and MRI-Based Forecasting}}\par\vspace{2pt}
\begin{cvitems}
  \item Worked on an Attention U-Net with temporal embedding, and Neural ODE models for continuous-time
        glioblastoma forecasting from longitudinal multimodal MRI.
  \item Reduced prediction MSE by 6.7\% versus a persistence baseline and relative FLAIR-volume
        error by 98\%; presented at the 2025 CMU Biomedical Engineering Forum.
\end{cvitems}

\section{Select Projects}
\cvProject{\href{https://github.com/tushar-nayak/cardiac-reconstruction-evolved}{Sparse Echocardiography to 3D Cardiac Volume Reconstruction}}{Aug 2025 -- May 2026}
\begin{cvitems}
  \item Reconstructed dense 3D left-ventricular occupancy from three 2D echocardiography slices
        using a neural implicit representation with Fourier features, differentiable projection and learnable SE(3) slice-pose refinement. 
  \item 
        Achieved 0.864 Dice and 0.765 IoU on \textsc{MITEA-CAP}. Also explored Gaussian occupancy fields (preliminary 0.950 IoU).
\end{cvitems}

\cvProject{\href{https://github.com/tushar-nayak/endo-splat}{Deformable 3D Reconstruction of Endoscopic Surgical Scenes}}{Apr 2026--Present}
\begin{cvitems}
  \item Prototype deformable 3D Gaussian Splatting for monocular endoscopy with camera-aware
        rendering, depth/pose initialization, deformation fields, and semantic feature alignment.
\end{cvitems}

\cvProject{\href{https://github.com/tushar-nayak/grading-cbisddsm}{Geometric Multi-View Breast Cancer Detection}}{Jan--May 2026}
\begin{cvitems}
  \item Built CC/MLO mammography alignment and cross-view attention for lesion localization and classification while leading upstream integration of segmentation and object detection models.
  \item Achieved reduced lesion-centroid discrepancy by 99.1\% and $\approx 0.8 $ classification accuracy and F1. 
\end{cvitems}
